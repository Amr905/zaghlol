String m(dynamic x) => x == null ? '-' : '${x.minor / 100} ${x.currency.code}';

void main() {
  final insts = {for (final f in Directory('$reg/institutions').listSync().whereType<File>()) f.uri.pathSegments.last: f.readAsStringSync()};
  final res = RegistryLoader.load(manifest: File('$reg/registry_manifest.json').readAsStringSync(), shared: File('$reg/shared.json').readAsStringSync(), institutions: insts);
  final pipe = ParsingPipeline(res.registry!);
  for (final e in layouts.entries) {
    final r = pipe.parseOcr(e.value.trim(), DateTime.utc(2026, 11, 25, 20));
    final s = switch (r) {
      ParseParsed p => 'FOUND ${m(p.fields.amount)} merchant="${p.fields.merchant?.display}" date=${p.fields.date?.toIso8601String().substring(0, 10)} via ${p.templateId} conf=${p.confidencePermille}',
      ParseUnrecognized u => 'NO TOTAL (${u.reason.name})',
      _ => r.runtimeType.toString(),
    };
    print('${e.key}: $s');
  }
}
